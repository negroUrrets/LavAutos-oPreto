/*
    DataBaseLavadero - Esquema de base de datos
    Motor: Microsoft SQL Server

    Descripcion:
        Crea la base de datos DataBaseLavadero con las tablas del dominio del
        lavadero de vehiculos, sus claves primarias, claves foraneas y relaciones.

    Alcance:
        Esquema y datos iniciales de los catalogos: utilerias, categorias y estados
        de lavado, categorias, marcas y modelos de vehiculo. Las inserciones son
        idempotentes y no destructivas.

    Caracteristicas:
        - Idempotente y no destructivo: crea la base y las tablas solo si no
          existen. No elimina objetos ni datos.
        - La herencia Cliente/Persona se implementa como tabla por tipo (TPT):
          Cliente comparte la clave primaria codigoPersona de Persona.
        - Se preserva la relacion Lavado -> Vehiculo -> Cliente, sin una clave
          foranea directa de Lavado hacia Cliente.

    Uso:
        Revisar el script y ejecutarlo manualmente en SQL Server Management
        Studio o sqlcmd sobre una instancia de SQL Server accesible.
*/

SET NOCOUNT ON;
GO

===== INICIO ARCHIVO: src/backend/config/database.js =====
const { Sequelize } = require('sequelize');
const { loadConfig } = require('./index');

const config = loadConfig();

function buildDialectOptions() {
  const options = {
    encrypt: config.db.encrypt,
    trustServerCertificate: config.db.trustServerCertificate,
  };

  if (config.db.instance) {
    options.instanceName = config.db.instance;
  }

  const dialectOptions = { options };

  if (config.db.authType === 'ntlm') {
    dialectOptions.authentication = {
      type: 'ntlm',
      options: {
        domain: config.db.domain,
        userName: config.db.user,
        password: config.db.password,
      },
    };
  }

  return dialectOptions;
}

const sequelize = new Sequelize(config.db.name, config.db.user, config.db.password, {
  host: config.db.server,
  port: config.db.instance ? undefined : config.db.port || 1433,
  dialect: config.db.dialect,
  dialectOptions: buildDialectOptions(),
  logging: false,
});

module.exports = { sequelize, config };

const REQUIRED_ENV = ['DB_SERVER', 'DB_NAME', 'DB_USER', 'DB_PASSWORD', 'SESSION_SECRET'];

function toBoolean(value, fallback) {
  if (value === undefined || value === '') {
    return fallback;
  }
  return value === 'true';
}

function loadConfig() {
  const missing = REQUIRED_ENV.filter((key) => !process.env[key]);
  if (missing.length > 0) {
    throw new Error(`Faltan variables de entorno requeridas: ${missing.join(', ')}`);
  }

  return {
    port: Number(process.env.PORT || 3000),
    db: {
      server: process.env.DB_SERVER,
      instance: process.env.DB_INSTANCE || undefined,
      name: process.env.DB_NAME,
      dialect: process.env.DB_DIALECT || 'mssql',
      port: process.env.DB_PORT ? Number(process.env.DB_PORT) : undefined,
      authType: (process.env.DB_AUTH_TYPE || 'default').toLowerCase(),
      user: process.env.DB_USER,
      password: process.env.DB_PASSWORD,
      domain: process.env.DB_DOMAIN || undefined,
      encrypt: toBoolean(process.env.DB_ENCRYPT, true),
      trustServerCertificate: toBoolean(process.env.DB_TRUST_SERVER_CERTIFICATE, true),
    },
    session: {
      secret: process.env.SESSION_SECRET,
      cookieSecure: toBoolean(process.env.SESSION_COOKIE_SECURE, false),
    },
  };
}

const AppError = require('../utils/AppError');

function normalizeError(err) {
  if (err instanceof AppError) {
    return err;
  }

  switch (err.name) {
    case 'SequelizeValidationError':
      return new AppError('Datos invalidos.', 400);
    case 'SequelizeUniqueConstraintError':
      return new AppError('El recurso ya existe.', 409);
    case 'SequelizeForeignKeyConstraintError':
      return new AppError('No se puede completar la operacion por dependencias existentes.', 409);
    default:
      return null;
  }
}

// eslint-disable-next-line no-unused-vars
function errorHandler(err, req, res, next) {
  const normalized = normalizeError(err);
  const statusCode = normalized ? normalized.statusCode : 500;
  const message = normalized ? normalized.message : 'Error interno del servidor';

  res.status(statusCode).json({ success: false, error: { message } });
}

module.exports = errorHandler;

const { Usuario } = require('../models');
const { hashPassword } = require('../utils/password');

const DEFAULT_ADMIN_USER = 'admin';
const DEFAULT_ADMIN_PASSWORD = '123456';

async function ensureDefaultAdmin({
  nombreUsuario = DEFAULT_ADMIN_USER,
  contrasena = DEFAULT_ADMIN_PASSWORD,
} = {}) {
  const existing = await Usuario.findOne({ where: { nombreUsuario } });
  if (existing) {
    return { created: false, codigoUsuario: existing.codigoUsuario };
  }

  const created = await Usuario.create({
    nombreUsuario,
    'contraseña': hashPassword(contrasena),
  });

  return { created: true, codigoUsuario: created.codigoUsuario };

const { Usuario, Login } = require('../models');
const AppError = require('../utils/AppError');
const { verifyPassword } = require('../utils/password');

const GENERIC_MESSAGE = 'Usuario o contrasena incorrectos.';

async function login(payload = {}, session) {
  const { usuario, contrasena } = payload;

  if (!usuario || !contrasena) {
    throw new AppError(GENERIC_MESSAGE, 401);
  }

  const user = await Usuario.findOne({ where: { nombreUsuario: usuario } });
  const passwordMatches = user ? verifyPassword(contrasena, user.get('contraseña')) : false;

  if (!user || !passwordMatches) {
    throw new AppError(GENERIC_MESSAGE, 401);
  }

  const loginRecord = await Login.create({ usuario: user.codigoUsuario });

  session.codigoLogin = loginRecord.codigoLogin;
  session.codigoUsuario = user.codigoUsuario;
  session.nombreUsuario = user.nombreUsuario;

  return {
    codigoUsuario: user.codigoUsuario,
    nombreUsuario: user.nombreUsuario,
    codigoLogin: loginRecord.codigoLogin,
  };
}

function logout(session) {
  return new Promise((resolve, reject) => {
    session.destroy((error) => (error ? reject(error) : resolve()));
  });
}

function getSession(session) {
  if (!session || !session.codigoUsuario) {
    throw new AppError('No hay sesion activa.', 401);
  }

  return {
    authenticated: true,
    codigoUsuario: session.codigoUsuario,
    nombreUsuario: session.nombreUsuario,
    codigoLogin: session.codigoLogin,
  };
}

const CATEGORIAS = ['Basico', 'Completo', 'Premium'];
const ESTADOS = ['En Espera', 'Lavando', 'Terminado', 'Entregado'];
const UTILERIAS = ['Auxilio', 'Balizas', 'Matafuegos'];

async function ensureRow(model, where, defaults) {
  const existing = await model.findOne({ where });
  if (existing) {
    return false;
  }
  await model.create({ ...where, ...defaults });
  return true;
}

async function ensureLavadoCatalogs() {
  let created = 0;

  for (const descripcionLavado of CATEGORIAS) {
    if (await ensureRow(CategoriaLavado, { descripcionLavado })) {
      created += 1;
    }
  }

  for (const descripcionEstado of ESTADOS) {
    if (await ensureRow(EstadoLavado, { descripcionEstado }, { estadoLavado: true })) {
      created += 1;
    }
  }

  for (const descripcionUtileria of UTILERIAS) {
    if (await ensureRow(Utileria, { descripcionUtileria }, { estadoUtileria: false })) {
      created += 1;
    }
  }

  return created;
}

const { Op } = require('sequelize');
const { sequelize, Persona, Cliente, Vehiculo, Lavado } = require('../models');
const AppError = require('../utils/AppError');
const { requireFields } = require('../utils/validators');
const { assertSinLavadoActivo } = require('./lavadoActivoService');

const FECHA_PATTERN = /^\d{4}-\d{2}-\d{2}$/;

function toClienteResponse(cliente) {
  const plain = cliente.get({ plain: true });
  const persona = plain.persona || {};

  return {
    codigoPersona: plain.codigoPersona,
    nombre: persona.nombre,
    apellido: persona.apellido,
    dni: persona.dni,
    email: persona.email,
    celular: persona.celular,
    estadoCliente: plain.estadoCliente,
  };
}

async function list(query = {}) {
  const personaInclude = { model: Persona, as: 'persona' };
  const term = typeof query.buscar === 'string' ? query.buscar.trim() : '';
  const sinLavadosDesde =
    typeof query.sinLavadosDesde === 'string' ? query.sinLavadosDesde.trim() : '';

  if (term) {
    const like = `%${term}%`;
    personaInclude.where = {
      [Op.or]: [
        { nombre: { [Op.like]: like } },
        { apellido: { [Op.like]: like } },
        sequelize.where(sequelize.cast(sequelize.col('persona.dni'), 'NVARCHAR(20)'), {
          [Op.like]: like,
        }),
      ],
    };
    personaInclude.required = true;
  }

  const where = { estadoCliente: true };

  if (sinLavadosDesde) {
    const excluir = await clientesConLavadosDesde(sinLavadosDesde);
    if (excluir.length > 0) {
      where.codigoPersona = { [Op.notIn]: excluir };
    }
  }

  const clientes = await Cliente.findAll({
    where,
    include: [personaInclude],
  });
  return clientes.map(toClienteResponse);
}

async function clientesConLavadosDesde(valor) {
  if (!FECHA_PATTERN.test(valor)) {
    throw new AppError('La fecha debe tener formato YYYY-MM-DD.', 400);
  }
  const limite = new Date(`${valor}T23:59:59.999`);
  if (Number.isNaN(limite.getTime())) {
    throw new AppError('La fecha no es valida.', 400);
  }

  const lavados = await Lavado.findAll({
    where: { fechaHoraLavado: { [Op.gt]: limite } },
    include: [{ model: Vehiculo, as: 'vehiculo', attributes: ['cliente'] }],
  });

  return [...new Set(lavados.map((lavado) => lavado.vehiculo?.cliente).filter(Boolean))];
}

async function getById(id) {
  const cliente = await Cliente.findByPk(id, {
    include: [{ model: Persona, as: 'persona' }],
  });
  if (!cliente) {
    throw new AppError('Cliente no encontrado.', 404);
  }
  return toClienteResponse(cliente);
}

async function create(payload) {
  requireFields(payload, ['nombre', 'apellido', 'dni']);
  const estadoCliente = payload.estadoCliente === undefined ? true : payload.estadoCliente;

  return sequelize.transaction(async (transaction) => {
    const persona = await Persona.create(
      {
        nombre: payload.nombre,
        apellido: payload.apellido,
        dni: payload.dni,
        email: payload.email ?? null,
        celular: payload.celular ?? null,
      },
      { transaction },
    );

    await Cliente.create({ codigoPersona: persona.codigoPersona, estadoCliente }, { transaction });

    const created = await Cliente.findByPk(persona.codigoPersona, {
      include: [{ model: Persona, as: 'persona' }],
      transaction,
    });
    return toClienteResponse(created);
  });
}

async function update(id, payload) {
  return sequelize.transaction(async (transaction) => {
    const cliente = await Cliente.findByPk(id, { transaction });
    if (!cliente) {
      throw new AppError('Cliente no encontrado.', 404);
    }

    const vehiculosDelCliente = await Vehiculo.findAll({
      where: { cliente: id },
      attributes: ['codigoVehi'],
      transaction,
    });
    await assertSinLavadoActivo(
      vehiculosDelCliente.map((vehiculo) => vehiculo.codigoVehi),
      transaction,
    );

    const persona = await Persona.findByPk(id, { transaction });
    await persona.update(
      {
        nombre: payload.nombre ?? persona.nombre,
        apellido: payload.apellido ?? persona.apellido,
        dni: payload.dni ?? persona.dni,
        email: payload.email !== undefined ? payload.email : persona.email,
        celular: payload.celular !== undefined ? payload.celular : persona.celular,
      },
      { transaction },
    );

    if (payload.estadoCliente !== undefined) {
      await cliente.update({ estadoCliente: payload.estadoCliente }, { transaction });
    }

    const updated = await Cliente.findByPk(id, {
      include: [{ model: Persona, as: 'persona' }],
      transaction,
    });
    return toClienteResponse(updated);
  });
}

async function remove(id) {
  return sequelize.transaction(async (transaction) => {
    const cliente = await Cliente.findByPk(id, { transaction });
    if (!cliente) {
      throw new AppError('Cliente no encontrado.', 404);
    }

    const vehiculos = await Vehiculo.findAll({ where: { cliente: id }, transaction });
    const vehiculoIds = vehiculos.map((vehiculo) => vehiculo.codigoVehi);
    await assertSinLavadoActivo(vehiculoIds, transaction);

    const lavados = vehiculoIds.length
      ? await Lavado.count({ where: { vehi: { [Op.in]: vehiculoIds } }, transaction })
      : 0;

    if (lavados > 0) {
      await cliente.update({ estadoCliente: false }, { transaction });
      await Vehiculo.update({ estadoVehi: false }, { where: { cliente: id }, transaction });
      return;
    }

    await Vehiculo.destroy({ where: { cliente: id }, transaction });
    await cliente.destroy({ transaction });
    await Persona.destroy({ where: { codigoPersona: id }, transaction });
  });
}

module.exports = { list, getById, create, update, remove };

const AppError = require('../utils/AppError');

function pickFields(payload, fields) {
  if (!fields) {
    return payload;
  }

  return fields.reduce((acc, field) => {
    if (payload[field] !== undefined) {
      acc[field] = payload[field];
    }
    return acc;
  }, {});
}

function createCrudService({
  model,
  name,
  fields,
  validateCreate,
  validateUpdate,
  buildListOptions,
  include,
}) {
  const baseOptions = () => (include ? { include } : {});

  async function list(query = {}) {
    const listOptions = { ...baseOptions(), ...(buildListOptions ? buildListOptions(query) : {}) };
    return model.findAll(listOptions);
  }

  async function getById(id) {
    const record = await model.findByPk(id, baseOptions());
    if (!record) {
      throw new AppError(`${name} no encontrado.`, 404);
    }
    return record;
  }

  async function create(payload) {
    const data = pickFields(payload, fields);
    if (validateCreate) {
      await validateCreate(data);
    }
    return model.create(data);
  }

  async function update(id, payload) {
    const record = await getById(id);
    const data = pickFields(payload, fields);
    if (validateUpdate) {
      await validateUpdate(data);
    }
    await record.update(data);
    return record;
  }

  async function remove(id) {
    const record = await getById(id);
    await record.destroy();
  }

  return { list, getById, create, update, remove };
}

module.exports = { createCrudService, pickFields };

    where: {
      vehi: { [Op.in]: ids },
      estado: { [Op.in]: estados.map((estado) => estado.codigoEstado) },
    },
    transaction,
  });

  if (cantidad > 0) {
    throw new AppError('Vehiculo en estado de Lavado.', 409);
  }
}

===== INICIO ARCHIVO: src/backend/utils/password.js =====
const crypto = require('crypto');

const KEY_LENGTH = 64;
const PREFIX = 'scrypt';

function hashPassword(password) {
  const salt = crypto.randomBytes(16).toString('hex');
  const derived = crypto.scryptSync(password, salt, KEY_LENGTH).toString('hex');
  return `${PREFIX}$${salt}$${derived}`;
}

function verifyPassword(password, stored) {
  if (typeof stored !== 'string') {
    return false;
  }

  const parts = stored.split('$');
  if (parts.length !== 3 || parts[0] !== PREFIX) {
    return false;
  }

  const [, salt, hash] = parts;
  const derived = crypto.scryptSync(password, salt, KEY_LENGTH).toString('hex');
  const expected = Buffer.from(hash, 'hex');
  const actual = Buffer.from(derived, 'hex');

  if (expected.length !== actual.length) {
    return false;
  }

  return crypto.timingSafeEqual(expected, actual);
}

module.exports = { hashPassword, verifyPassword };

const AppError = require('./AppError');

function isMissing(value) {
  return value === undefined || value === null || value === '';
}

function requireFields(payload, fields) {
  const missing = fields.filter((field) => isMissing(payload[field]));
  if (missing.length > 0) {
    throw new AppError(`Faltan campos obligatorios: ${missing.join(', ')}`, 400);
  }
}

async function assertExists(model, id, message) {
  if (isMissing(id)) {
    throw new AppError(message, 400);
  }
  const found = await model.findByPk(id);
  if (!found) {
    throw new AppError(message, 400);
  }
}

module.exports = { requireFields, assertExists };

      </div>

        <LogoutPopup onConfirm={handleConfirmLogout} onClose={() => setShowLogout(false)} />
      ) : null}

  );
