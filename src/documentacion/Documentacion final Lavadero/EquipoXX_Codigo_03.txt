const ESTADOS = [
  { codigoEstado: 1, descripcionEstado: 'En Espera' },
  { codigoEstado: 2, descripcionEstado: 'Lavando' },

];

let sessionOk = true;
let lavados = [];

function data(value) {
  return Promise.resolve({ ok: true, json: () => Promise.resolve({ data: value }) });
}

function crearLavado(codigoLavado, codigoEstado) {
  return {
    codigoLavado,
    fechaHoraLavado: '2026-09-07T13:05:00Z',
    estado: codigoEstado,
    datosEstado: ESTADOS.find((estado) => estado.codigoEstado === codigoEstado),
    categoria: { descripcionLavado: 'Completo' },
    vehiculo: {
      patenteVehi: `AB-12${codigoLavado}-CD`,
      modelo: { modeloVehi: 'Corsa', marca: { marcaVehi: 'Chevrolet' } },
      datosCliente: {
        persona: {
          dni: 30111222,
          nombre: 'Ana',
          apellido: 'Perez',
          celular: '1155512345',
          email: 'ana@correo.com',
        },
      },
    },
  };
}

function mockFetch(url, options = {}) {
  if (url === '/api/auth/session') {
    return Promise.resolve({
      ok: sessionOk,
      status: sessionOk ? 200 : 401,
      json: () => Promise.resolve({ data: sessionOk ? { codigoLogin: 1 } : null }),
    });
  }
  if (url === '/api/lavados') {
    return data(lavados);
  }
  if (url === '/api/estados-lavado') {
    return data(ESTADOS);
  }
  if (typeof url === 'string' && url.startsWith('/api/lavados/') && options.method === 'PUT') {
    const id = Number(url.split('/').pop());
    const body = JSON.parse(options.body);
    lavados = lavados.map((lavado) =>
      lavado.codigoLavado === id
        ? {
            ...lavado,
            estado: body.estado,
            datosEstado: ESTADOS.find(
              (estado) => String(estado.codigoEstado) === String(body.estado),
            ),
          }
        : lavado,
    );
    return data({ codigoLavado: id, estado: body.estado });
  }
  return data([]);
}

  );
}

import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import autoIcon from '../assets/auto.png';
import bubblesIcon from '../assets/burbujas.png';
import Popup from '../components/Popup.jsx';
import { login } from '../services/authService.js';

const SUCCESS_MESSAGE = '!Bienvenid@!';
const ERROR_MESSAGE = 'Usuario / Contraseña incorrecta';

export default function LoginPage() {
  const [usuario, setUsuario] = useState('');
  const [contrasena, setContrasena] = useState('');
  const [fieldError, setFieldError] = useState('');
  const [popupMessage, setPopupMessage] = useState('');
  const [isSuccess, setIsSuccess] = useState(false);
  const navigate = useNavigate();

  async function handleSubmit(event) {
    event.preventDefault();
    setFieldError('');

    if (!usuario.trim() || !contrasena.trim()) {
      setFieldError('Usuario y contraseña son obligatorios.');
      return;
    }

    try {
      const ok = await login(usuario, contrasena);
      setIsSuccess(ok);
      setPopupMessage(ok ? SUCCESS_MESSAGE : ERROR_MESSAGE);
    } catch {
      setPopupMessage(ERROR_MESSAGE);
    }
  }

  return (
    <div className="flex min-h-screen items-center justify-center bg-brand-indigo  p-4">
      <div className="w-full max-w-md rounded-3xl bg-brand-pink border-2 border-gray-400 px-10 py-12 shadow-2xl">
        <div className="flex flex-col items-center">
          <img src={bubblesIcon} alt="" className="h-8 w-14" />
          <img src={autoIcon} alt="Auto de lavadero" className="h-14 w-18" />
        </div>

        <h1 className="mt-4 text-center text-4xl font-extrabold text-brand-title">
          LavAutos oPreto
        </h1>

        <form className="mt-10 space-y-6" onSubmit={handleSubmit} noValidate>
          <div>
            <label htmlFor="usuario" className="block text-lg font-bold text-brand-navy">
              Usuario
            </label>
            <input
              id="usuario"
              name="usuario"
              type="text"
              value={usuario}
              onChange={(event) => setUsuario(event.target.value)}
              className="mt-2 w-full rounded-full bg-brand-input px-5 py-3 text-white outline-none"
            />
          </div>

          <div>
            <label htmlFor="contrasena" className="block text-lg font-bold text-brand-navy">
              Contraseña
            </label>
            <input
              id="contrasena"
              name="contrasena"
              type="password"
              value={contrasena}
              onChange={(event) => setContrasena(event.target.value)}
              className="mt-2 w-full rounded-full bg-brand-input px-5 py-3 text-white outline-none"
            />
          </div>

          {fieldError ? (
            <p role="alert" className="text-sm font-bold text-red-700">
              {fieldError}
            </p>
          ) : null}

          {/* <button
            type="submit"
            className=" ml-30  rounded-full bg-brand-green py-3 text-xl font-bold text-white"
          >
            Iniciar Sesion
          </button> */}

          <button
  type="submit"
  className="mx-auto block px-12 py-3 text-xl font-bold text-white rounded-full bg-brand-green"
>
  Iniciar Sesion
</button>

 
        </form>
      </div>

      {popupMessage ? (
        <Popup
          message={popupMessage}
          onClose={() => {
            setPopupMessage('');
            if (isSuccess) {
              navigate('/home');
            }
          }}
        />
      ) : null}
    </div>
  );
}

import { useEffect, useState } from 'react';
import { useNavigate } from 'react-router-dom';
import autoIcon from '../assets/auto.png';
import userIcon from '../assets/usuario.png';
import Popup from '../components/Popup.jsx';
import { fetchSession } from '../services/authService.js';
import { createCliente, searchClientes } from '../services/clientService.js';
import {
  createVehiculo,
  listCategorias,
  listMarcas,
  listModelos,
} from '../services/vehicleService.js';
import {
  formatPatente,
  isValidEmail,
  isValidPatente,
  onlyDigits,
  onlyLetters,
} from '../utils/validators.js';

const EMPTY_FORM = {
  dni: '',
  nombre: '',
  apellido: '',
  celular: '',
  email: '',
  categoria: '',
  marca: '',
  modelo: '',
  patente: '',
};

export default function NuevoClientePage() {
  const navigate = useNavigate();
  const [checking, setChecking] = useState(true);
  const [form, setForm] = useState(EMPTY_FORM);
  const [categorias, setCategorias] = useState([]);
  const [marcas, setMarcas] = useState([]);
  const [modelos, setModelos] = useState([]);
  const [warning, setWarning] = useState('');
  const [popup, setPopup] = useState(null);

  useEffect(() => {
    let active = true;

    fetchSession()
      .then((ok) => {
        if (!active) {
          return;
        }
        if (!ok) {
          navigate('/login', { replace: true });
        } else {
          setChecking(false);
        }
      })
      .catch(() => {
        if (active) {
          navigate('/login', { replace: true });
        }
      });

    return () => {
      active = false;
    };
  }, [navigate]);

  useEffect(() => {
    listCategorias().then(setCategorias);
    listMarcas().then(setMarcas);
    listModelos().then(setModelos);
  }, []);

  function updateField(name, value) {
    setForm((prev) => {
      const next = { ...prev, [name]: value };
      if (name === 'categoria') {
        next.marca = '';
        next.modelo = '';
      }
      if (name === 'marca') {
        next.modelo = '';
      }
      return next;
    });
  }

  const marcasFiltradas = marcas.filter(
    (marca) => String(marca.categoriaVehi) === String(form.categoria),
  );
  const modelosFiltrados = modelos.filter(
    (modelo) => String(modelo.marcaVehi) === String(form.marca),
  );

  async function handleSubmit(event) {
    event.preventDefault();
    setWarning('');

    const incompleto = Object.values(form).some((value) => !String(value).trim());
    if (incompleto) {
      setWarning('Todos los campos son obligatorios.');
      return;
    }

    if (!isValidEmail(form.email)) {
      setWarning('El e-mail no tiene un formato válido.');
      return;
    }

    if (!isValidPatente(form.patente)) {
      setWarning('La patente debe tener el formato AA-123-AA.');
      return;
    }

    const dni = String(form.dni).trim();
    const existentes = await searchClientes(dni);
    const yaExiste = existentes.some((cliente) => String(cliente.dni) === dni);

    if (yaExiste) {
      setPopup({ message: 'Ese cliente ya se encuentra registrado.', action: 'clear' });
      return;
    }

    try {
      const cliente = await createCliente({
        nombre: form.nombre,
        apellido: form.apellido,
        dni: form.dni,
        email: form.email,
        celular: form.celular,
      });
      await createVehiculo({
        patenteVehi: form.patente,
        estadoVehi: true,
        cliente: cliente.codigoPersona,
        modeloVehi: form.modelo,
      });
      setPopup({ message: `Bienvenido ${form.nombre}`, action: 'navigate' });
    } catch {
      setWarning('No se pudo registrar el cliente.');
    }
  }

  function handlePopupClose() {
    const action = popup?.action;
    setPopup(null);
    if (action === 'clear') {
      setForm(EMPTY_FORM);
    } else if (action === 'navigate') {
      navigate('/clientes');
    }
  }

  if (checking) {
    return null;
  }

  const inputClass =
    'mt-1 w-full rounded-full bg-brand-input px-4 py-2 text-white outline-none';
  const labelClass = 'block text-sm font-bold text-white';

  return (
    <div className="flex min-h-screen items-center justify-center bg-brand-indigo p-6">
      <div className="w-full max-w-md rounded-3xl bg-brand-pink p-6">
        <h1 className="text-2xl font-extrabold text-[#5ea014]">Nuevo Cliente</h1>

        <form onSubmit={handleSubmit} noValidate>
          <div className="mt-4 rounded-2xl bg-brand-indigo p-4">
            <div className="mb-3 flex justify-center">
              <img src={userIcon} alt="Usuario" className="h-12 w-12" />
            </div>
            <div className="space-y-3">
              <div>
                <label htmlFor="dni" className={labelClass}>
                  DNI
                </label>
                <input
                  id="dni"
                  name="dni"
                  type="text"
                  value={form.dni}
                  onChange={(event) => updateField('dni', onlyDigits(event.target.value))}
                  className={inputClass}
                />
              </div>
              <div>
                <label htmlFor="nombre" className={labelClass}>
                  Nombre
                </label>
                <input
                  id="nombre"
                  name="nombre"
                  type="text"
                  value={form.nombre}
                  onChange={(event) => updateField('nombre', onlyLetters(event.target.value))}
                  className={inputClass}
                />
              </div>
              <div>
                <label htmlFor="apellido" className={labelClass}>
                  Apellido
                </label>
                <input
                  id="apellido"
                  name="apellido"
                  type="text"
                  value={form.apellido}
                  onChange={(event) => updateField('apellido', onlyLetters(event.target.value))}
                  className={inputClass}
                />
              </div>
              <div>
                <label htmlFor="celular" className={labelClass}>
                  Celular
                </label>
                <input
                  id="celular"
                  name="celular"
                  type="text"
                  value={form.celular}
                  onChange={(event) => updateField('celular', onlyDigits(event.target.value))}
                  className={inputClass}
                />
              </div>
              <div>
                <label htmlFor="email" className={labelClass}>
                  e-mail
                </label>
                <input
                  id="email"
                  name="email"
                  type="text"
                  value={form.email}
                  onChange={(event) => updateField('email', event.target.value)}
                  className={inputClass}
                />

function codigo(valor) {
  return String(valor);
}

export default function NuevoLavadoPage() {
  const { id, vehiculoId } = useParams();
  const navigate = useNavigate();
  const [checking, setChecking] = useState(true);
  const [vehiculo, setVehiculo] = useState(null);
  const [utilerias, setUtilerias] = useState([]);
  const [categorias, setCategorias] = useState([]);
  const [estados, setEstados] = useState([]);
  const [seleccion, setSeleccion] = useState([]);
  const [categoria, setCategoria] = useState('');
  const [estado, setEstado] = useState('');
  const [detalles, setDetalles] = useState('');
  const [login, setLogin] = useState(null);
  const [warning, setWarning] = useState('');
  const [showConfirm, setShowConfirm] = useState(false);

  useEffect(() => {
    let active = true;

    fetchSessionData()
      .then((session) => {
        if (!active) {
          return;
        }
        if (!session) {
          navigate('/login', { replace: true });
          return;
        }
        setLogin(session.codigoLogin);

        Promise.all([
          getVehiculo(vehiculoId),
          listUtileria(),
          listCategoriasLavado(),
          listEstadosLavado(),
        ]).then(([datosVehiculo, listaUtileria, listaCategorias, listaEstados]) => {
          if (!active) {
            return;
          }
          setVehiculo(datosVehiculo);
          setUtilerias(listaUtileria);
          setCategorias(listaCategorias);
          setEstados(listaEstados);
          setSeleccion(
            listaUtileria
              .filter((utileria) => Boolean(utileria.estadoUtileria))
              .map((utileria) => codigo(utileria.codigoUtileria)),
          );
          setChecking(false);
        });
      })
      .catch(() => {
        if (active) {
          navigate('/login', { replace: true });
        }
      });

    return () => {
      active = false;
    };
  }, [vehiculoId, navigate]);

  async function toggleUtileria(codigoUtileria) {
    const activa = seleccion.includes(codigoUtileria);
    const nuevoEstado = !activa;

    try {
      await updateUtileria(codigoUtileria, { estadoUtileria: nuevoEstado });
      setSeleccion((prev) =>
        nuevoEstado ? [...prev, codigoUtileria] : prev.filter((item) => item !== codigoUtileria),
      );
      setWarning('');
    } catch {
      setWarning('No se pudo actualizar la utileria.');
    }
  }

  function handleSubmit(event) {
    event.preventDefault();
    setWarning('');

    if (!categoria) {
      setWarning('Seleccione una categoria.');
      return;
    }
    if (!estado) {
      setWarning('Seleccione un estado.');
      return;
    }
    if (seleccion.length === 0) {
      setWarning('Seleccione al menos una utileria.');
      return;
    }

    setShowConfirm(true);
  }

  async function handleConfirm() {
    try {
      await createLavado({
        vehi: vehiculoId,
        categoriaLavado: categoria,
        estado,
        utileria: seleccion[0],
        detallesLavado: detalles,
        login,
      });

      navigate(`/clientes/${id}/info`);
    } catch {
      setShowConfirm(false);

    }
  }

  function handleCancelConfirm() {

                      className={`h-6 w-12 rounded-full p-1 transition-colors ${

const EMPTY_FORM = {
  categoria: '',
  marca: '',
  modelo: '',
  patente: '',
};

export default function NuevoVehiculoPage() {
  const { id } = useParams();
  const navigate = useNavigate();
  const [checking, setChecking] = useState(true);
  const [form, setForm] = useState(EMPTY_FORM);
  const [categorias, setCategorias] = useState([]);
  const [marcas, setMarcas] = useState([]);
  const [modelos, setModelos] = useState([]);
  const [warning, setWarning] = useState('');
  const [popup, setPopup] = useState(null);

  useEffect(() => {
    let active = true;

    fetchSession()
      .then((ok) => {
        if (!active) {
          return;
        }
        if (!ok) {
          navigate('/login', { replace: true });
        } else {
          setChecking(false);
        }
      })
      .catch(() => {
        if (active) {
          navigate('/login', { replace: true });
        }
      });

    return () => {
      active = false;
    };
  }, [navigate]);

  useEffect(() => {
    listCategorias().then(setCategorias);
    listMarcas().then(setMarcas);
    listModelos().then(setModelos);
  }, []);

  function updateField(name, value) {
    setForm((prev) => {
      const next = { ...prev, [name]: value };
      if (name === 'categoria') {
        next.marca = '';
        next.modelo = '';
      }
      if (name === 'marca') {
        next.modelo = '';
      }
      return next;
    });
  }

  const marcasFiltradas = marcas.filter(
    (marca) => String(marca.categoriaVehi) === String(form.categoria),
  );
  const modelosFiltrados = modelos.filter(
    (modelo) => String(modelo.marcaVehi) === String(form.marca),
  );

  async function handleSubmit(event) {
    event.preventDefault();
    setWarning('');

    const incompleto = Object.values(form).some((value) => !String(value).trim());
    if (incompleto) {
      setWarning('Todos los campos son obligatorios.');
      return;
    }

    if (!isValidPatente(form.patente)) {
      setWarning('La patente debe tener el formato AA-123-AA.');
      return;
    }

    const patente = form.patente.trim().toUpperCase();
    const existentes = await listVehiculos();
    const yaExiste = existentes.some(
      (vehiculo) => String(vehiculo.patenteVehi).trim().toUpperCase() === patente,
    );

    if (yaExiste) {
      setPopup({ message: 'Vehiculo ya registrado :/', action: 'clear' });
      return;
    }

    try {
      await createVehiculo({
        patenteVehi: form.patente,
        estadoVehi: true,
        cliente: id,
        modeloVehi: form.modelo,

      setWarning('No se pudo registrar el vehiculo.');
    }

  function handlePopupClose() {
    const action = popup?.action;
    setPopup(null);
    if (action === 'clear') {
      setForm(EMPTY_FORM);
    } else if (action === 'navigate') {
      navigate(`/clientes/${id}/info`);
    }
  }

  if (checking) {
    return null;
  }

  const inputClass =

  const labelClass = 'block text-sm font-bold text-white';

  return (

            <div className="mb-3 flex justify-center">
              <img src={autoIcon} alt="Vehiculo" className="h-10 w-10" />
            </div>
            <div className="space-y-3">
              <div>
                <label htmlFor="categoria" className={labelClass}>
                  Vehiculo
                </label>
                <select
                  id="categoria"
                  name="categoria"
                  value={form.categoria}
                  onChange={(event) => updateField('categoria', event.target.value)}
                  className={inputClass}
                >
                  <option value="">Seleccionar...</option>
                  {categorias.map((categoria) => (
                    <option key={categoria.codigoCategoriaVehi} value={categoria.codigoCategoriaVehi}>
                      {categoria.categoriaVehi}
                    </option>
                  ))}
                </select>
              </div>
              <div>
                <label htmlFor="marca" className={labelClass}>
                  Marca
                </label>
                <select
                  id="marca"
                  name="marca"
                  value={form.marca}
                  onChange={(event) => updateField('marca', event.target.value)}
                  className={inputClass}
                >
                  <option value="">Seleccionar...</option>
                  {marcasFiltradas.map((marca) => (
                    <option key={marca.codigoMarcaVehi} value={marca.codigoMarcaVehi}>
                      {marca.marcaVehi}
                    </option>
                  ))}
                </select>
              </div>
              <div>
                <label htmlFor="modelo" className={labelClass}>
                  Modelo

                  value={form.modelo}

                  <option value="">Seleccionar...</option>
                  {modelosFiltrados.map((modelo) => (
                    <option key={modelo.codigoModeloVehi} value={modelo.codigoModeloVehi}>

    navigate(`/clientes/${cliente?.codigoPersona}/editar`);
  }

  async function handleEliminarCliente() {
    if (await clienteBloqueado()) {
      setShowBloqueo(true);
      return;
    }
    setShowDeleteCliente(true);
  }

  async function handleEditarVehiculo(item) {
    if (await tieneLavadoActivo([item.codigoVehi])) {
      setShowBloqueo(true);
      return;
    }
    navigate(`/clientes/${cliente?.codigoPersona}/vehiculos/${item.codigoVehi}/editar`);
  }

  async function handleEliminarVehiculo(item) {
    if (await tieneLavadoActivo([item.codigoVehi])) {
      setShowBloqueo(true);
      return;
    }
    setShowDeleteVehiculo(true);
  }

  async function handleConfirmDeleteCliente() {
    try {
      await deleteCliente(cliente?.codigoPersona);
      navigate('/clientes');
    } catch (error) {
      setShowDeleteCliente(false);
      if (error.status === 409) {
        setShowBloqueo(true);
        return;
      }
      navigate('/clientes');
    }
  }

  async function handleConfirmDeleteVehiculo() {
    try {
      await deleteVehiculo(vehiculoId);
      navigate('/vehiculos');
    } catch (error) {
      setShowDeleteVehiculo(false);
      if (error.status === 409) {
        setShowBloqueo(true);
        return;
      }
      navigate('/vehiculos');
    }
  }

  if (checking) {
    return null;
  }

  return (
    <>
      <ClienteVehiculoInfo
        cliente={cliente}
        vehiculos={vehiculo ? [vehiculo] : []}
        onNuevoVehiculo={() => navigate(`/clientes/${cliente?.codigoPersona}/vehiculos/nuevo`)}
        onEditarCliente={handleEditarCliente}
        onEliminarCliente={handleEliminarCliente}
        onEditarVehiculo={handleEditarVehiculo}
        onEliminarVehiculo={handleEliminarVehiculo}
        onVolver={() => navigate('/vehiculos')}
      />

      {showDeleteCliente ? (
        <ConfirmPopup
          message={`¿ Desea eliminar al cliente "${cliente?.nombre ?? ''} ${cliente?.apellido ?? ''}" ?`}
          onConfirm={handleConfirmDeleteCliente}
          onCancel={() => setShowDeleteCliente(false)}
        />
      ) : null}

      {showDeleteVehiculo ? (
        <ConfirmPopup
          message={`¿ Desea eliminar el vehiculo "${vehiculo?.patenteVehi ?? ''}" ?`}
          onConfirm={handleConfirmDeleteVehiculo}
          onCancel={() => setShowDeleteVehiculo(false)}
        />
      ) : null}

      {showBloqueo ? (
        <Popup message={MENSAJE_LAVADO_ACTIVO} onClose={() => setShowBloqueo(false)} />
      ) : null}
    </>
  );
}

function matches(vehiculo, termino) {
  const texto = termino.trim().toUpperCase();
  if (!texto) {
    return false;
  }
  const patente = String(vehiculo.patenteVehi ?? '').toUpperCase();
  const modelo = String(vehiculo.modelo?.modeloVehi ?? '').toUpperCase();
  return patente.includes(texto) || modelo.includes(texto);
}

export default function VehiculosPage() {
  const navigate = useNavigate();

  useEffect(() => {
    let active = true;

    fetchSession()
      .then((ok) => {
        if (!active) {
          return;
        }
        if (!ok) {
          navigate('/login', { replace: true });
          return;
        }
        listVehiculos().then((lista) => {
          if (active) {
            setVehiculos(lista);
            setChecking(false);
          }
        });

      .catch(() => {
        if (active) {
          navigate('/login', { replace: true });
        }

      active = false;

    return null;

export async function tieneLavadoActivo(vehiculoIds = []) {
  const ids = vehiculoIds.filter((id) => id !== undefined && id !== null);

  if (ids.length === 0) {
    return false;
  }

  const listas = await Promise.all(ids.map((id) => listByVehiculo(id)));

  return listas.some((lavados) =>
    lavados.some((lavado) =>
      ESTADOS_BLOQUEANTES.includes(normalizarEstado(lavado.datosEstado?.descripcionEstado)),
    ),
  );
}

export function listLavados(filtros = {}) {
  const params = new URLSearchParams();

      params.set(campo, filtros[campo]);
    }
  });
  const query = params.toString();
  return listResource(query ? `/api/lavados?${query}` : '/api/lavados');
}

const TIME_ZONE = 'America/Argentina/Buenos_Aires';

function capitalize(text) {
  return text.charAt(0).toUpperCase() + text.slice(1);
}

function partsToMap(parts) {
  return parts.reduce((acc, part) => {
    acc[part.type] = part.value;
    return acc;
  }, {});
}

export function formatDate(date) {
  const parts = new Intl.DateTimeFormat('es-AR', {
    timeZone: TIME_ZONE,
    weekday: 'long',
    day: '2-digit',
    month: '2-digit',
    year: 'numeric',
  }).formatToParts(date);
  const map = partsToMap(parts);

  return `${capitalize(map.weekday || '')} ${map.day}/${map.month}/${map.year}`;
}

export function formatTime(date) {
  const parts = new Intl.DateTimeFormat('es-AR', {
    timeZone: TIME_ZONE,
    hour: '2-digit',
    minute: '2-digit',
    hour12: true,
  }).formatToParts(date);
  const map = partsToMap(parts);
  const period = (map.dayPeriod || '').replace(/[^a-zA-Z]/g, '').toLowerCase() || 'am';

  return { hour: map.hour || '', minute: map.minute || '', period };
}

const EMAIL_PATTERN = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const PATENTE_PATTERN = /^[A-Z]{2}-\d{3}-[A-Z]{2}$/;
const PATENTE_TYPES = ['L', 'L', 'N', 'N', 'N', 'L', 'L'];

export function isValidEmail(texto) {
  return EMAIL_PATTERN.test(String(texto).trim());
}

export function isValidPatente(texto) {
  return PATENTE_PATTERN.test(String(texto).trim());
}

export function formatPatente(texto) {
  const raw = String(texto)
    .toUpperCase()
    .replace(/[^A-Z0-9]/g, '');

  let value = '';
  let index = 0;

  for (const char of raw) {
    if (index >= PATENTE_TYPES.length) {
      break;
    }

    const expected = PATENTE_TYPES[index];
    const isLetter = char >= 'A' && char <= 'Z';
    const isNumber = char >= '0' && char <= '9';

    if ((expected === 'L' && isLetter) || (expected === 'N' && isNumber)) {
      value += char;
      index += 1;
    }
  }

  const letters1 = value.slice(0, 2);
  const numbers = value.slice(2, 5);
  const letters2 = value.slice(5, 7);

  let formatted = letters1;
  if (value.length > 2) {
    formatted += `-${numbers}`;
  }
  if (value.length > 5) {
    formatted += `-${letters2}`;
  }

  return formatted;
}

export function onlyDigits(texto) {
  return String(texto).replace(/\D/g, '');
}

export function onlyLetters(texto) {
  return String(texto).replace(/[^A-Za-zÁÉÍÓÚÜÑáéíóúüñ]/g, '');

import { describe, expect, test } from 'vitest';
import {
  formatPatente,
  isValidEmail,
  isValidPatente,
  onlyDigits,
  onlyLetters,
} from './validators.js';

describe('validators', () => {
  test('isValidEmail acepta un correo válido y rechaza uno inválido', () => {
    expect(isValidEmail('fernando@gmail.com')).toBe(true);
    expect(isValidEmail('fernando@')).toBe(false);
    expect(isValidEmail('fernando')).toBe(false);
  });

  test('formatPatente aplica la máscara', () => {
    expect(formatPatente('ab254mc')).toBe('AB-254-MC');
    expect(formatPatente('AB2')).toBe('AB-2');
    expect(formatPatente('AB254')).toBe('AB-254');
  });

  test('formatPatente descarta el tipo incorrecto', () => {
    expect(formatPatente('A1B254MC')).toBe('AB-254-MC');
    expect(formatPatente('1')).toBe('');
  });

  test('isValidPatente acepta el formato completo y rechaza incompletos', () => {
    expect(isValidPatente('AB-254-MC')).toBe(true);
    expect(isValidPatente('AB-254-M')).toBe(false);
    expect(isValidPatente('A-254-MC')).toBe(false);
  });

  test('onlyDigits conserva solo números', () => {
    expect(onlyDigits('12a3')).toBe('123');
    expect(onlyDigits('a-b c')).toBe('');
  });

  test('onlyLetters conserva solo letras con acentos y ñ', () => {
    expect(onlyLetters('Muñoz 1')).toBe('Muñoz');
    expect(onlyLetters('Ángel')).toBe('Ángel');
    expect(onlyLetters('A1-B')).toBe('AB');
  });
});
