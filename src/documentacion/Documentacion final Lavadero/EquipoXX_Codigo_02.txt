    <div className="min-h-screen bg-brand-indigo p-6">

    <div className="flex flex-1 flex-col items-center justify-center rounded-3xl bg-brand-panel px-10 py-14 text-center text-white">

        <span className="text-3xl font-bold">  {minute}</span>
        <span className="ml-2 text-2xl font-bold">{period}</span>

  );

import { useEffect, useState } from 'react';
import { useLocation, useNavigate, useParams } from 'react-router-dom';
import ClienteVehiculoInfo from '../components/ClienteVehiculoInfo.jsx';
import ConfirmPopup from '../components/ConfirmPopup.jsx';
import Popup from '../components/Popup.jsx';
import { fetchSession } from '../services/authService.js';
import { deleteCliente, getCliente } from '../services/clientService.js';
import { MENSAJE_LAVADO_ACTIVO, tieneLavadoActivo } from '../services/lavadoService.js';
import { deleteVehiculo, listByCliente } from '../services/vehicleService.js';

export default function ClienteInfoPage() {
  const { id } = useParams();
  const navigate = useNavigate();
  const location = useLocation();
  const origen = location.state?.from ?? '/clientes';
  const [checking, setChecking] = useState(true);
  const [cliente, setCliente] = useState(null);
  const [vehiculos, setVehiculos] = useState([]);
  const [showDelete, setShowDelete] = useState(false);
  const [showDeleteVehiculo, setShowDeleteVehiculo] = useState(null);
  const [showLavarWarning, setShowLavarWarning] = useState(false);
  const [showBloqueo, setShowBloqueo] = useState(false);

  useEffect(() => {
    let active = true;

    fetchSession()
      .then((ok) => {
        if (!active) {
          return;
        }
        if (!ok) {
          navigate('/login', { replace: true });
          return;
        }
        Promise.all([getCliente(id), listByCliente(id)]).then(([datosCliente, listaVehiculos]) => {
          if (!active) {
            return;
          }
          setCliente(datosCliente);
          setVehiculos(listaVehiculos);
          setChecking(false);
        });
      })
      .catch(() => {
        if (active) {
          navigate('/login', { replace: true });
        }
      });

    return () => {
      active = false;
    };
  }, [id, navigate]);

  async function handleConfirmDelete() {
    try {
      await deleteCliente(id);
      navigate('/clientes');
    } catch (error) {
      setShowDelete(false);
      if (error.status === 409) {
        setShowBloqueo(true);
        return;
      }
      navigate('/clientes');
    }
  }

  async function handleConfirmDeleteVehiculo() {
    try {
      await deleteVehiculo(showDeleteVehiculo?.codigoVehi);
      const lista = await listByCliente(id);
      setVehiculos(lista);
      setShowDeleteVehiculo(null);
      navigate(`/clientes/${id}/info`, { replace: true });
    } catch (error) {
      setShowDeleteVehiculo(null);
      if (error.status === 409) {
        setShowBloqueo(true);
        return;
      }
      navigate(`/clientes/${id}/info`, { replace: true });
    }
  }

  function handleCancelDeleteVehiculo() {
    setShowDeleteVehiculo(null);
    navigate(`/clientes/${id}/info`, { replace: true });
  }

  async function handleEditarCliente() {
    if (await tieneLavadoActivo(vehiculos.map((vehiculo) => vehiculo.codigoVehi))) {
      setShowBloqueo(true);
      return;
    }
    navigate(`/clientes/${id}/editar`);
  }

  async function handleEliminarCliente() {
    if (await tieneLavadoActivo(vehiculos.map((vehiculo) => vehiculo.codigoVehi))) {
      setShowBloqueo(true);
      return;
    }
    setShowDelete(true);
  }

  async function handleEditarVehiculo(vehiculo) {
    if (await tieneLavadoActivo([vehiculo.codigoVehi])) {
      setShowBloqueo(true);
      return;
    }
    navigate(`/clientes/${id}/vehiculos/${vehiculo.codigoVehi}/editar`);
  }

  async function handleEliminarVehiculo(vehiculo) {
    if (await tieneLavadoActivo([vehiculo.codigoVehi])) {
      setShowBloqueo(true);
      return;
    }
    setShowDeleteVehiculo(vehiculo);
  }

  async function handleLavar(vehiculo) {
    const bloqueado = await tieneLavadoActivo([vehiculo.codigoVehi]);

    if (bloqueado) {
      setShowLavarWarning(true);
      return;
    }

    navigate(`/clientes/${id}/vehiculos/${vehiculo.codigoVehi}/lavar`);
  }

  if (checking) {
    return null;
  }

  return (
    <>
      <ClienteVehiculoInfo
        cliente={cliente}
        vehiculos={vehiculos}
        onNuevoVehiculo={() => navigate(`/clientes/${id}/vehiculos/nuevo`)}
        onEditarCliente={handleEditarCliente}
        onEliminarCliente={handleEliminarCliente}
        onEditarVehiculo={handleEditarVehiculo}
        onEliminarVehiculo={handleEliminarVehiculo}
        onLavar={handleLavar}
        onVolver={() => navigate(origen)}
      />

      {showDelete ? (
        <ConfirmPopup
          message={`¿ Desea eliminar al cliente "${cliente?.nombre ?? ''} ${cliente?.apellido ?? ''}" ?`}
          onConfirm={handleConfirmDelete}
          onCancel={() => setShowDelete(false)}
        />
      ) : null}

      {showDeleteVehiculo ? (
        <ConfirmPopup
          message={`¿ Desea eliminar el vehiculo "${showDeleteVehiculo.patenteVehi ?? ''}" ?`}
          onConfirm={handleConfirmDeleteVehiculo}
          onCancel={handleCancelDeleteVehiculo}
        />
      ) : null}

      {showLavarWarning ? (
        <Popup
          message="El vehiculo ya tiene un lavado en curso."
          onClose={() => setShowLavarWarning(false)}
        />
      ) : null}

      {showBloqueo ? (
        <Popup message={MENSAJE_LAVADO_ACTIVO} onClose={() => setShowBloqueo(false)} />
      ) : null}
    </>
  );
}

function mockFetch(url, options = {}) {
  if (url === '/api/auth/session') {
    return Promise.resolve({ ok: sessionOk, status: sessionOk ? 200 : 401 });
  }
  if (url === '/api/clientes/5' && options.method === 'DELETE') {
    return Promise.resolve({ ok: deleteStatus < 400, status: deleteStatus });
  }
  if (url === '/api/clientes/5') {
    return Promise.resolve({ ok: true, json: () => Promise.resolve({ data: cliente }) });
  }
  if (url === '/api/vehiculos/9' && options.method === 'DELETE') {
    if (deleteStatus < 400) {
      vehiculos = vehiculos.filter((vehiculo) => vehiculo.codigoVehi !== 9);
    }
    return Promise.resolve({ ok: deleteStatus < 400, status: deleteStatus });
  }
  if (url === '/api/vehiculos?cliente=5') {
    return Promise.resolve({ ok: true, json: () => Promise.resolve({ data: vehiculos }) });
  }
  if (url === '/api/lavados?vehi=9') {
    return Promise.resolve({ ok: true, json: () => Promise.resolve({ data: lavadosVehiculo }) });
  }
  return Promise.resolve({ ok: true, json: () => Promise.resolve({ data: [] }) });
}

const EMPTY_FORM = { dni: '', nombre: '', apellido: '', celular: '', email: '' };

function toForm(cliente) {
  return {
    dni: cliente?.dni != null ? String(cliente.dni) : '',
    nombre: cliente?.nombre || '',
    apellido: cliente?.apellido || '',
    celular: cliente?.celular != null ? String(cliente.celular) : '',
    email: cliente?.email || '',
  };
}

export default function EditarClientePage() {
  const { id } = useParams();
  const navigate = useNavigate();
  const [checking, setChecking] = useState(true);
  const [form, setForm] = useState(EMPTY_FORM);
  const [original, setOriginal] = useState(EMPTY_FORM);
  const [warning, setWarning] = useState('');
  const [popup, setPopup] = useState(null);

  useEffect(() => {
    let active = true;

    fetchSession()
      .then((ok) => {
        if (!active) {
          return;
        }
        if (!ok) {
          navigate('/login', { replace: true });
          return;
        }
        getCliente(id).then((cliente) => {
          if (!active) {
            return;
          }
          const datos = toForm(cliente);
          setOriginal(datos);
          setForm(datos);
          setChecking(false);
        });
      })
      .catch(() => {
        if (active) {
          navigate('/login', { replace: true });
        }
      });

    return () => {
      active = false;
    };
  }, [id, navigate]);

  function updateField(name, value) {
    setForm((prev) => ({ ...prev, [name]: value }));
  }

  async function handleSubmit(event) {
    event.preventDefault();
    setWarning('');

    const incompleto = Object.values(form).some((value) => !String(value).trim());
    if (incompleto) {
      setWarning('Todos los campos son obligatorios.');
      return;
    }

    if (!isValidEmail(form.email)) {
      setWarning('El e-mail no tiene un formato válido.');
      return;
    }

    const dni = String(form.dni).trim();
    const existentes = await searchClientes(dni);
    const duplicado = existentes.some(
      (cliente) => String(cliente.dni) === dni && String(cliente.codigoPersona) !== String(id),
    );

    if (duplicado) {
      setPopup({ message: 'Ese cliente ya se encuentra registrado.', action: 'reset' });
      return;
    }

    try {
      await updateCliente(id, {
        nombre: form.nombre,
        apellido: form.apellido,
        dni: form.dni,
        email: form.email,
        celular: form.celular,
      });
      setPopup({ message: 'La edición de datos se efectuó satisfactoriamente.', action: 'info' });
    } catch {
      setWarning('No se pudo guardar la edición.');
    }
  }

  function handlePopupClose() {
    const action = popup?.action;
    setPopup(null);
    if (action === 'reset') {
      setForm(original);
    } else if (action === 'info') {
      navigate(`/clientes/${id}/info`);
    }
  }

  if (checking) {
    return null;
  }

                  DNI
                </label>

                  name="dni"
                  type="text"
                  value={form.dni}
                  onChange={(event) => updateField('dni', onlyDigits(event.target.value))}
                  className={inputClass}
                />
              </div>

                  className={inputClass}
                />

              <div>
                <label htmlFor="apellido" className={labelClass}>

                </label>
                <input
                  id="apellido"

import { useEffect, useState } from 'react';
import { useNavigate, useParams } from 'react-router-dom';
import BrandHeader from '../components/BrandHeader.jsx';
import Popup from '../components/Popup.jsx';
import { fetchSession } from '../services/authService.js';
import {
  getLavado,
  listCategoriasLavado,
  listEstadosLavado,
  listUtileria,
  updateLavado,
  updateUtileria,
} from '../services/lavadoService.js';

function codigo(valor) {
  return String(valor);
}

export default function EditarLavadoPage() {
  const { lavadoId } = useParams();
  const navigate = useNavigate();
  const [checking, setChecking] = useState(true);
  const [lavado, setLavado] = useState(null);
  const [utilerias, setUtilerias] = useState([]);
  const [categorias, setCategorias] = useState([]);
  const [estados, setEstados] = useState([]);
  const [seleccion, setSeleccion] = useState([]);
  const [categoria, setCategoria] = useState('');
  const [estado, setEstado] = useState('');
  const [detalles, setDetalles] = useState('');
  const [warning, setWarning] = useState('');
  const [showSuccess, setShowSuccess] = useState(false);

  useEffect(() => {
    let active = true;

    fetchSession()
      .then((ok) => {
        if (!active) {
          return;
        }
        if (!ok) {
          navigate('/login', { replace: true });
          return;
        }

        Promise.all([
          getLavado(lavadoId),
          listUtileria(),
          listCategoriasLavado(),
          listEstadosLavado(),
        ])
          .then(([datosLavado, listaUtileria, listaCategorias, listaEstados]) => {
            if (!active) {
              return;
            }
            if (!datosLavado) {
              navigate('/lavados', { replace: true });
              return;
            }
            setLavado(datosLavado);
            setUtilerias(listaUtileria);
            setCategorias(listaCategorias);
            setEstados(listaEstados);
            setSeleccion(
              listaUtileria
                .filter((utileria) => Boolean(utileria.estadoUtileria))
                .map((utileria) => codigo(utileria.codigoUtileria)),
            );
            setCategoria(codigo(datosLavado.categoriaLavado));
            setEstado(codigo(datosLavado.estado));
            setDetalles(datosLavado.detallesLavado ?? '');
            setChecking(false);
          })
          .catch(() => {
            if (active) {
              navigate('/lavados', { replace: true });
            }
          });
      })
      .catch(() => {
        if (active) {
          navigate('/login', { replace: true });
        }
      });

    return () => {
      active = false;
    };
  }, [lavadoId, navigate]);

  async function toggleUtileria(codigoUtileria) {
    const activa = seleccion.includes(codigoUtileria);
    const nuevoEstado = !activa;

    try {
      await updateUtileria(codigoUtileria, { estadoUtileria: nuevoEstado });
      setSeleccion((prev) =>
        nuevoEstado ? [...prev, codigoUtileria] : prev.filter((item) => item !== codigoUtileria),
      );
      setWarning('');
    } catch {
      setWarning('No se pudo actualizar la utileria.');
    }
  }

  async function handleRegistrar() {
    try {
      await updateLavado(lavadoId, {
        categoriaLavado: categoria,
        estado,
        utileria: seleccion[0],
        detallesLavado: detalles,
      });
      setShowSuccess(true);
    } catch {
      setWarning('No se pudo editar el lavado.');
    }
  }

  function handleSubmit(event) {
    event.preventDefault();
    setWarning('');

    if (!categoria) {
      setWarning('Seleccione una categoria.');
      return;
    }
    if (!estado) {
      setWarning('Seleccione un estado.');
      return;
    }
    if (seleccion.length === 0) {
      setWarning('Seleccione al menos una utileria.');
      return;
    }

    handleRegistrar();
  }

  if (checking) {
    return null;
  }

  const labelClass = 'block text-sm font-bold text-white';

  return (
    <div className="min-h-screen bg-brand-indigo p-6">
      <BrandHeader />

      <div className="mx-auto mt-2 max-w-3xl rounded-3xl border-3 border-blue-900 bg-[#8f8a8a] p-6">
        <h1 className="text-xl font-bold text-brand-navy">
          Editar Lavado: <span className="font-extrabold">{lavado?.vehiculo?.patenteVehi}</span>
        </h1>

        <form onSubmit={handleSubmit} className="mt-4 rounded-2xl bg-[#2637a2] p-6">
          <div className="flex flex-wrap gap-8">
            <div className="flex-1 space-y-3">
              {utilerias.map((utileria) => {
                const valor = codigo(utileria.codigoUtileria);
                return (
                  <label
                    key={valor}
                    className="flex items-center gap-3 text-sm font-bold text-white"
                  >
                    <input
                      type="checkbox"
                      checked={seleccion.includes(valor)}
                      onChange={() => toggleUtileria(valor)}
                      className="h-5 w-5"
                    />
                    {utileria.descripcionUtileria}
                  </label>
                );
              })}
            </div>

            <div className="flex-1 space-y-3">
              {categorias.map((item) => {
                const valor = codigo(item.codigoCategoriaLavado);
                const activa = categoria === valor;
                return (
                  <div
                    key={valor}
                    className="flex items-center justify-between gap-3 text-sm font-bold text-white"
                  >
                    <span>{item.descripcionLavado}</span>
                    <button
                      type="button"
                      role="switch"
                      aria-checked={activa}
                      aria-label={item.descripcionLavado}
                      onClick={() => setCategoria(valor)}
                      className={`h-6 w-12 rounded-full p-1 transition-colors ${
                        activa ? 'bg-green-600' : 'bg-red-600'
                      }`}
                    >
                      <span
                        className={`block h-4 w-4 rounded-full bg-white transition-transform ${
                          activa ? 'translate-x-6' : 'translate-x-0'
                        }`}
                      />
                    </button>
                  </div>
                );
              })}
            </div>
          </div>

          <div className="mt-6">
            <label htmlFor="detalles" className={labelClass}>
              Detalles
            </label>
            <textarea
              id="detalles"
              value={detalles}
              onChange={(event) => setDetalles(event.target.value)}
              placeholder="Detalles..."
              rows={3}
              className="mt-1 w-full rounded-2xl bg-brand-input px-4 py-2 text-white outline-none"
            />
          </div>

          <div className="mt-6 flex justify-center">
            <div className="flex items-center gap-3 rounded-xl border border-white px-4 py-2">
              <label htmlFor="estado" className={labelClass}>
                Estado...
              </label>
              <select
                id="estado"
                value={estado}
                onChange={(event) => setEstado(event.target.value)}
                className="rounded-md bg-white px-3 py-1 text-brand-navy outline-none"
              >
                <option value="">Seleccionar...</option>
                {estados.map((item) => (
                  <option key={item.codigoEstado} value={item.codigoEstado}>
                    {item.descripcionEstado}
                  </option>
                ))}
              </select>
            </div>
          </div>

          {warning ? (
            <p role="alert" className="mt-3 text-center text-sm font-bold text-red-700">
              {warning}
            </p>
          ) : null}

          <div className="mt-6 flex justify-center gap-6">
            <button

        </form>
      </div>

        <Popup message="Lavado editado exitosamente" onClose={() => navigate('/lavados')} />

    marca: String(vehiculo?.modelo?.marca?.codigoMarcaVehi ?? ''),
    modelo: String(vehiculo?.modelo?.codigoModeloVehi ?? ''),
    patente: vehiculo?.patenteVehi || '',
  };
}

export default function EditarVehiculoPage() {
  const { id, vehiculoId } = useParams();
  const navigate = useNavigate();
  const [checking, setChecking] = useState(true);
  const [form, setForm] = useState(EMPTY_FORM);
  const [original, setOriginal] = useState(EMPTY_FORM);
  const [categorias, setCategorias] = useState([]);
  const [marcas, setMarcas] = useState([]);
  const [modelos, setModelos] = useState([]);
  const [warning, setWarning] = useState('');
  const [popup, setPopup] = useState(null);

  useEffect(() => {
    let active = true;

    fetchSession()
      .then((ok) => {
        if (!active) {
          return;
        }
        if (!ok) {
          navigate('/login', { replace: true });
          return;
        }
        getVehiculo(vehiculoId).then((vehiculo) => {
          if (!active) {
            return;
          }
          const datos = toForm(vehiculo);
          setOriginal(datos);
          setForm(datos);
          setChecking(false);
        });
      })
      .catch(() => {
        if (active) {
          navigate('/login', { replace: true });
        }
      });

    return () => {
      active = false;
    };
  }, [vehiculoId, navigate]);

  useEffect(() => {
    listCategorias().then(setCategorias);
    listMarcas().then(setMarcas);
    listModelos().then(setModelos);
  }, []);

  function updateField(name, value) {
    setForm((prev) => {
      const next = { ...prev, [name]: value };
      if (name === 'categoria') {
        next.marca = '';
        next.modelo = '';
      }
      if (name === 'marca') {
        next.modelo = '';
      }
      return next;
    });
  }

  const marcasFiltradas = marcas.filter(
    (marca) => String(marca.categoriaVehi) === String(form.categoria),
  );
  const modelosFiltrados = modelos.filter(
    (modelo) => String(modelo.marcaVehi) === String(form.marca),
  );

  async function handleSubmit(event) {
    event.preventDefault();
    setWarning('');

    const incompleto = Object.values(form).some((value) => !String(value).trim());
    if (incompleto) {
      setWarning('Todos los campos son obligatorios.');
      return;
    }

    if (!isValidPatente(form.patente)) {
      setWarning('La patente debe tener el formato AA-123-AA.');
      return;
    }

    const patente = form.patente.trim().toUpperCase();
    const existentes = await listVehiculos();
    const duplicado = existentes.some(
      (vehiculo) =>
        String(vehiculo.codigoVehi) !== String(vehiculoId) &&
        String(vehiculo.patenteVehi).trim().toUpperCase() === patente,
    );

    if (duplicado) {
      setPopup({ message: 'Error: vehiculo ya registrado', action: 'reset' });
      return;
    }

    try {
      await updateVehiculo(vehiculoId, {
        patenteVehi: form.patente,
        estadoVehi: true,
        cliente: id,
        modeloVehi: form.modelo,
      });
      setPopup({ message: 'El vehiculo se edito correctamente.', action: 'info' });
    } catch {
      setWarning('No se pudo guardar la edición.');
    }
  }

  function handlePopupClose() {
    const action = popup?.action;
    setPopup(null);
    if (action === 'reset') {
      setForm(original);
    } else if (action === 'info') {
      navigate(`/clientes/${id}/info`);    
    }
  }

  if (checking) {
    return null;
  }

  const inputClass =
    'mt-1 w-full rounded-full bg-brand-input px-4 py-2 text-white outline-none';
  const labelClass = 'block text-sm font-bold text-white';

  return (
    <div className="flex min-h-screen items-center justify-center bg-brand-indigo p-6">
      <div className="w-full max-w-md rounded-3xl bg-brand-pink p-6">
        <h1 className="text-2xl font-extrabold text-[#7eb93c]">Editar Vehiculo</h1>

        <form onSubmit={handleSubmit} noValidate>
          <div className="mt-4 rounded-2xl bg-brand-indigo p-4">
            <div className="mb-3 flex justify-center">
              <img src={autoIcon} alt="Vehiculo" className="h-10 w-10" />
            </div>
            <div className="space-y-3">
              <div>
                <label htmlFor="categoria" className={labelClass}>
                  Vehiculo
                </label>
                <select
                  id="categoria"
                  name="categoria"
                  value={form.categoria}
                  onChange={(event) => updateField('categoria', event.target.value)}
                  className={inputClass}
                >
                  <option value="">Seleccionar...</option>
                  {categorias.map((categoria) => (
                    <option key={categoria.codigoCategoriaVehi} value={categoria.codigoCategoriaVehi}>
                      {categoria.categoriaVehi}
                    </option>
                  ))}
                </select>
              </div>
              <div>
                <label htmlFor="marca" className={labelClass}>
                  Marca
                </label>
                <select
                  id="marca"
                  name="marca"
                  value={form.marca}
                  onChange={(event) => updateField('marca', event.target.value)}
                  className={inputClass}
                >
                  <option value="">Seleccionar...</option>
                  {marcasFiltradas.map((marca) => (
                    <option key={marca.codigoMarcaVehi} value={marca.codigoMarcaVehi}>
                      {marca.marcaVehi}
                    </option>
                  ))}
                </select>
              </div>
              <div>
                <label htmlFor="modelo" className={labelClass}>
                  Modelo
                </label>

                  value={form.modelo}
                  onChange={(event) => updateField('modelo', event.target.value)}
                  className={inputClass}
                >
                  <option value="">Seleccionar...</option>
                  {modelosFiltrados.map((modelo) => (
                    <option key={modelo.codigoModeloVehi} value={modelo.codigoModeloVehi}>
                      {modelo.modeloVehi}
                    </option>
                  ))}
                </select>
              </div>
              <div>
                <label htmlFor="patente" className={labelClass}>
                  Patente
                </label>
                <input
                  id="patente"
                  name="patente"
                  type="text"
                  value={form.patente}
                  onChange={(event) => updateField('patente', formatPatente(event.target.value))}
                  className={inputClass}
                />
              </div>
            </div>
          </div>

          {warning ? (
            <p role="alert" className="mt-3 text-sm font-bold text-red-700">
              {warning}
            </p>
          ) : null}

          <div className="mt-6 flex justify-center gap-6">
            <button
              type="submit"

              className="rounded-full bg-red-600 px-8 py-2 font-bold text-white"

            </button>
          </div>
        </form>
      </div>

    </div>
  );
}

import LavadoInfoPopup from '../components/LavadoInfoPopup.jsx';
import Popup from '../components/Popup.jsx';
import { fetchSession } from '../services/authService.js';
import { listEstadosLavado, listLavados, updateLavado } from '../services/lavadoService.js';
import { formatTime } from '../utils/formatDateTime.js';

const SECCIONES = [
  {
    id: 'en-espera',
    titulo: 'En Espera...',
    estados: ['en espera'],
    borde: 'border-red-600',
    etiqueta: 'text-red-700',
    accion: { etiqueta: 'Lavar', popup: 'Lavando Vehiculo', destino: 'Lavando' },
  },
  {
    id: 'lavando',
    titulo: 'Lavando...',
    estados: ['lavando'],
    borde: 'border-yellow-700',
    etiqueta: 'text-yellow-700',
    accion: { etiqueta: 'Completado', popup: 'Lavado Terminado', destino: 'Terminado' },
  },
  {
    id: 'terminado',
    titulo: 'Terminado...',
    estados: ['terminado'],
    borde: 'border-green-600',
    etiqueta: 'text-green-700',
    accion: { etiqueta: 'Entregar', popup: 'Vehiculo Entregado', destino: 'Entregado' },
  },
];

function normalizar(texto) {
  return String(texto ?? '')
    .trim()
    .toLowerCase();
}

function formatHora(valor) {
  if (!valor) {
    return '';
  }
  const fecha = new Date(valor);
  if (Number.isNaN(fecha.getTime())) {
    return '';
  }
  const { hour, minute, period } = formatTime(fecha);
  return `${hour}:${minute} ${period}`;
}

function ChevronIcon({ arriba }) {
  return (
    <svg
      viewBox="0 0 24 24"
      aria-hidden="true"
      className="h-4 w-4"
      fill="none"
      stroke="currentColor"
      strokeWidth="3"
      strokeLinecap="round"
      strokeLinejoin="round"
    >
      <path d={arriba ? 'M6 15l6-6 6 6' : 'M6 9l6 6 6-6'} />
    </svg>
  );
}

function HandIcon() {
  return (
    <svg viewBox="0 0 24 24" aria-hidden="true" className="h-6 w-6" fill="currentColor">
      <path d="M8.5 12.5V6a1.5 1.5 0 0 1 3 0v4.5H12V5a1.5 1.5 0 0 1 3 0v5.5h.5V7a1.5 1.5 0 0 1 3 0v7.5a6.5 6.5 0 0 1-6.5 6.5h-.5A6.5 6.5 0 0 1 5 14.5v-3a1.5 1.5 0 0 1 3 0v1h.5z" />
    </svg>
  );
}

export default function LavadosPage() {
  const navigate = useNavigate();
  const [checking, setChecking] = useState(true);
  const [lavados, setLavados] = useState([]);
  const [estados, setEstados] = useState([]);
  const [colapsadas, setColapsadas] = useState({});
  const [info, setInfo] = useState(null);
  const [avance, setAvance] = useState(null);

  const cargar = useCallback(async () => {
    const [listaLavados, catalogoEstados] = await Promise.all([
      listLavados(),
      listEstadosLavado(),
    ]);
    setLavados(listaLavados);
    setEstados(catalogoEstados);
  }, []);

  useEffect(() => {
    let active = true;

    fetchSession()
      .then((ok) => {
        if (!active) {
          return;
        }
        if (!ok) {
          navigate('/login', { replace: true });
          return;
        }
        cargar()
          .then(() => {
            if (active) {
              setChecking(false);
            }
          })
          .catch(() => {
            if (active) {
              setChecking(false);
            }
          });
      })
      .catch(() => {
        if (active) {
          navigate('/login', { replace: true });
        }
      });

    return () => {
      active = false;
    };
  }, [cargar, navigate]);

  async function handleAcceptAvance() {
    if (!avance) {
      return;
    }
    const { lavado, seccion } = avance;
    const destino = estados.find(
      (estado) => normalizar(estado.descripcionEstado) === normalizar(seccion.accion.destino),
    );
    setAvance(null);

    if (!destino) {
      return;
    }

    await updateLavado(lavado.codigoLavado, { estado: destino.codigoEstado });
    await cargar();
  }

  function lavadosDe(seccion) {
    return lavados.filter((lavado) =>
      seccion.estados.includes(normalizar(lavado.datosEstado?.descripcionEstado)),
    );
  }

  function toggleSeccion(id) {
    setColapsadas((prev) => ({ ...prev, [id]: !prev[id] }));
  }

  if (checking) {
    return null;
  }

  return (
    <div className="min-h-screen bg-brand-indigo p-6">
      <BrandHeader />

      <div className="mt-4 flex justify-center">
        <button
          type="button"
          onClick={() => navigate('/clientes')}
          className="animate-nuevo-lavado flex items-center gap-2 rounded-full bg-green-600 px-8 py-2 font-bold text-white hover:[animation-play-state:paused]"
        >
          <HandIcon />
          Nuevo Lavado
        </button>
      </div>

      <div className="mx-auto mt-8 max-w-3xl space-y-8">
        {SECCIONES.map((seccion) => {
          const items = lavadosDe(seccion);
          const colapsada = Boolean(colapsadas[seccion.id]);

          return (
            <section key={seccion.id} aria-label={seccion.titulo}>
              <div className="flex items-center gap-3">
                <img
                  src={bubblesIcon}
                  alt=""
                  aria-hidden="true"
                  className="animate-burbuja h-10 w-10"
                />
                <div
                  className={`flex flex-1 items-center gap-3 rounded-full border-2 ${seccion.borde} bg-brand-input px-4 py-1`}
                >
                  <span className={`text-sm font-bold ${seccion.etiqueta}`}>{seccion.titulo}</span>
                  <button
                    type="button"
                    onClick={() => toggleSeccion(seccion.id)}
                    aria-label={`${colapsada ? 'Expandir' : 'Contraer'} ${seccion.titulo}`}
                    className={`ml-auto flex h-6 w-6 items-center justify-center rounded-full border ${seccion.borde} ${seccion.etiqueta}`}
                  >
                    <ChevronIcon arriba={colapsada} />
                  </button>
                </div>
              </div>

              {colapsada ? null : (
                <div className="mt-3 space-y-4">
                  {items.map((lavado) => (
                    <div
                      key={lavado.codigoLavado}
                      className="mx-auto w-full max-w-2xl rounded-2xl border border-white bg-[#2637a2] p-6"
                    >
                      <div className="flex flex-wrap items-center gap-4">
                        <label className="flex items-center gap-2">
                          <span className="text-sm font-bold text-white">N° Lavado</span>
                          <input
                            readOnly
                            aria-label="N° Lavado"
                            value={lavado.codigoLavado ?? ''}
                            className="w-20 rounded-md bg-brand-pink px-2 py-1 text-sm text-brand-navy"
                          />
                        </label>
                        <label className="flex items-center gap-2">
                          <span className="text-sm font-bold text-white">Hora</span>
                          <input
                            readOnly
                            aria-label="Hora"
                            value={formatHora(lavado.fechaHoraLavado)}
                            className="w-28 rounded-md bg-brand-pink px-2 py-1 text-sm text-brand-navy"
                          />
                        </label>
                        <label className="flex items-center gap-2">
                          <span className="text-sm font-bold text-white">Patente</span>
                          <input
                            readOnly
                            aria-label="Patente"
                            value={lavado.vehiculo?.patenteVehi ?? ''}
                            className="w-28 rounded-md bg-brand-pink px-2 py-1 text-sm text-brand-navy"
                          />
                        </label>
                      </div>

                      <div className="mt-4 flex justify-end gap-2">
                        <button
                          type="button"
                          onClick={() => setInfo(lavado)}
                          className="rounded-full bg-brand-green px-4 py-1 text-sm font-bold text-white"
                        >
                          + info
                        </button>
                        <button
                          type="button"
                          onClick={() => navigate(`/lavados/${lavado.codigoLavado}/editar`)}
                          className="rounded-full bg-gray-400 px-4 py-1 text-sm font-bold text-white"
                        >
                          Editar
                        </button>
                      </div>

                      <div className="mt-3 flex justify-center">
                        <button
                          type="button"
                          onClick={() => setAvance({ lavado, seccion })}
                          className="rounded-full bg-green-600 px-8 py-2 font-bold text-white"
                        >
                          {seccion.accion.etiqueta}
                        </button>
                      </div>
                    </div>
                  ))}
                </div>
              )}
            </section>
          );
        })}
      </div>

      <div className="mt-8 flex justify-center">
        <button
          type="button"
          onClick={() => navigate('/home')}
          className="rounded-full bg-green-600 px-8 py-2 font-bold text-white"
        >
          ←
        </button>
      </div>

      {info ? <LavadoInfoPopup lavado={info} onClose={() => setInfo(null)} /> : null}

      {avance ? (
        <Popup message={avance.seccion.accion.popup} onClose={handleAcceptAvance} />
      ) : null}
    </div>
  );
}
